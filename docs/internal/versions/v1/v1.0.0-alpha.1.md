# v1.0.0-alpha.1 — Core Types

**Status:** Planned  
**Type:** Minor Changes  
**SemVer:** 1.0.0-alpha.1

## Overview

The first alpha release establishes the core types: `Result`, `Maybe`, and `Unit` with first-class support for `@deessejs/errors`. Every type supports a **dual API**: instance methods for data-first style, and pipeable static functions for data-last/pipe style.

## Design Principles

1. **Dual API** — Every type has both instance methods and pipeable static functions
2. **@deessejs/errors integration** — First-class support for structured errors
3. **Simple by default** — Basic usage should be intuitive without configuration
4. **Type-safe** — Full TypeScript inference without runtime overhead

## File Structure

Each type group gets its own folder with separation of concerns:

```
src/
├── result/
│   ├── types.ts        # Ok, Err, Result interfaces
│   ├── constants.ts    # ok(), err() constructors
│   ├── builders.ts     # (reserved for future)
│   └── index.ts
│
├── maybe/
│   ├── types.ts        # Some, None, Maybe interfaces
│   ├── constants.ts    # some(), none(), maybe() constructors
│   ├── builders.ts     # (reserved for future)
│   └── index.ts
│
├── unit/
│   ├── types.ts        # Unit type
│   ├── constants.ts    # unit constant, isUnit()
│   └── index.ts
│
├── types.ts            # OkType, ErrType, SomeType, isResult(), isMaybe()
│
└── index.ts           # barrel: Result, Maybe, Unit + all exports
```

| File | Purpose |
|------|---------|
| `types.ts` | Interface definitions, type aliases |
| `constants.ts` | Constructor functions, constants |
| `builders.ts` | Complex factory functions (reserved) |
| `index.ts` | Re-exports for the group |

## Changes

### Minor Changes

- Add `Result<T, E>` type with `Ok` and `Err` variants
- Add `Maybe<T>` type with `Some` and `None` variants
- Add `Unit` type for void-returning functions
- Add constructors: `ok()`, `err()`, `some()`, `none()`, `unit`
- Add dual API: instance methods + pipeable static functions
- Add type guards: `isOk`, `isErr`, `isSome`, `isNone`, `isUnit`
- Add `@deessejs/errors` integration via `mapError`

## What's Included

### Result Type

```typescript
type Result<T, E> = Ok<T, E> | Err<T, E>;

interface Ok<T, E = never> {
  readonly _tag: 'Ok';
  readonly value: T;
}

interface Err<T = never, E> {
  readonly _tag: 'Err';
  readonly error: E;
}
```

**Constructors:**

```typescript
// Data-last (pipeable)
ok<T>(value: T): Ok<T, never>;
err<E>(error: E): Err<never, E>;
```

**Instance Methods:**

```typescript
// Transform value
map<B>(fn: (value: T) => B): Result<B, E>;
flatMap<B, E2>(fn: (value: T) => Result<B, E2>): Result<B, E | E2>;

// Transform error (supports @deessejs/errors)
mapError<E2>(fn: (error: E) => E2): Result<T, E2>;

// Filter and tap
filter(predicate: (value: T) => boolean, errorFn?: (value: T) => E): Result<T, E>;
tap(fn: (value: T) => unknown): Result<T, E>;
tapAsync(fn: (value: T) => Promise<unknown>): Promise<Result<T, E>>;

// Async variants
flatMapAsync<B, E2>(fn: (value: T) => Promise<Result<B, E2>>): Promise<Result<B, E | E2>>;

// Pattern matching
match<U>(handlers: { ok: (value: T) => U; err: (error: E) => U }): U;
fold<U>(onOk: (value: T) => U, onErr: (error: E) => U): U;

// Extract value
getOrElse(defaultValue: T): T;
getOrThrow(message?: string): T;
getOrNull(): T | null;
getOrUndefined(): T | undefined;

// Conversions
toMaybe(): Maybe<T>;
toOption(): Maybe<T>; // alias for toMaybe

// Type guards
isOk(): this is Ok<T, E>;
isErr(): this is Err<T, E>;
```

**Pipeable Static Functions (Dual API):**

```typescript
import { pipe, map, flatMap, mapError, filter, tap, match } from '@deessejs/fp';

// Data-last style with pipe
const result = pipe(
  ok(10),
  map(x => x * 2),
  mapError(e => new Error(String(e))),
  filter(x => x > 0, () => new Error('Not positive')),
  tap(x => console.log(x)),
);

// Equivalent to:
const result2 = ok(10)
  .map(x => x * 2)
  .mapError(e => new Error(String(e)))
  .filter(x => x > 0, () => new Error('Not positive'))
  .tap(x => console.log(x));
```

### Maybe Type

```typescript
type Maybe<T> = Some<T> | None;

interface Some<T> {
  readonly _tag: 'Some';
  readonly value: T;
}

interface None {
  readonly _tag: 'None';
}
```

**Constructors:**

```typescript
// Data-last (pipeable)
some<T>(value: T): Some<T>;
const none: None;

// Also available as:
maybe(value: T | null | undefined): Maybe<T>;
```

**Instance Methods:**

```typescript
// Transform value
map<B>(fn: (value: T) => B): Maybe<B>;
flatMap<B>(fn: (value: T) => Maybe<B>): Maybe<B>;

// Filter
filter(predicate: (value: T) => boolean): Maybe<T>;
filterMap<B>(fn: (value: T) => Maybe<B>): Maybe<B>;

// Tap
tap(fn: (value: T) => unknown): Maybe<T>;
tapAsync(fn: (value: T) => Promise<unknown>): Promise<Maybe<T>>;

// Pattern matching
match<U>(handlers: { some: (value: T) => U; none: () => U }): U;
fold<U>(onSome: (value: T) => U, onNone: () => U): U;

// Extract value
getOrElse(defaultValue: T): T;
getOrThrow(message?: string): T;
getOrNull(): T | null;
getOrUndefined(): T | undefined;

// Safe property access
get<K extends keyof T>(key: K): Maybe<T[K]>;

// Conversions
toResult<E>(error: E): Result<T, E>;
toArray(): T[];
toIterable(): Iterable<T>;

 // Type guards
isSome(): this is Some<T>;
isNone(): this is None;
```

**Pipeable Static Functions (Dual API):**

```typescript
import { pipe, map, flatMap, filter, tap, match, some, none } from '@deessejs/fp';

// Data-last style
const value = pipe(
  maybe(user?.address?.city),
  filter(city => city.length > 0),
  map(city => city.toUpperCase()),
  getOrElse('Unknown'),
);

// Pattern matching
const greeting = match(
  maybeUser,
  {
    some: (user) => `Hello, ${user.name}`,
    none: () => 'Hello, guest',
  }
);
```

### Unit Type

```typescript
type Unit = { readonly _tag: 'Unit' };
const unit: Unit;

// Type guard
isUnit(value: unknown): value is Unit;
```

### Type Utilities

```typescript
// Check if value is Result
isResult(value: unknown): value is Result<unknown, unknown>;

// Check if value is Maybe
isMaybe(value: unknown): value is Maybe<unknown>;

// Extract Ok value type from Result
type OkType<R extends Result<unknown, unknown>> = R extends Ok<infer T, unknown> ? T : never;

// Extract Err type from Result
type ErrType<R extends Result<unknown, unknown>> = R extends Ok<unknown, infer E> ? E : never;

// Extract Some value type from Maybe
type SomeType<M extends Maybe<unknown>> = M extends Some<infer T> ? T : never;
```

## @deessejs/errors Integration

First-class support for structured errors from `@deessejs/errors`.

### Error Hierarchy Example

```typescript
import { Result, ok, err, mapError } from '@deessejs/fp';
import { error, is } from '@deessejs/errors';

// Define error hierarchy
const AppError = error({
  name: 'AppError',
  message: 'Application error: {reason}',
});

const ValidationError = AppError.child({
  name: 'ValidationError',
  message: 'Validation failed: {reason}',
});

const NotFoundError = AppError.child({
  name: 'NotFoundError',
  message: 'Resource "{id}" not found',
});

// Using with Result
async function getUser(id: string): Promise<Result<User, ValidationError | NotFoundError>> {
  if (!id) {
    return err(ValidationError({ reason: 'ID is required' }));
  }

  const user = await db.findUser(id);
  if (!user) {
    return err(NotFoundError({ id }));
  }

  return ok(user);
}

// Error handling with type guards
const result = await getUser('123');
if (result.isErr() && is(result.error, ValidationError)) {
  console.log(result.error.fields.reason); // Type-safe access
}
```

### Error Traversal

```typescript
import { Result, ok, err } from '@deessejs/fp';
import { error, causes } from '@deessejs/errors';

const DatabaseError = error({
  name: 'DatabaseError',
  message: 'Database error: {cause}',
});

async function fetchData() {
  return tryPromise(() => api.fetch())
    .mapError(e => DatabaseError({ cause: e }));
}

// Traverse error chain
const result = await fetchData();
if (result.isErr()) {
  for (const err of causes(result.error)) {
    console.log(`${err.name}: ${err.message}`);
  }
}
```

## Usage Examples

### Basic Result

```typescript
import { Result, ok, err, pipe, map, flatMap, match } from '@deessejs/fp';

// Data-first (instance methods)
const divide = (a: number, b: number): Result<number, string> =>
  b === 0 ? err('Division by zero') : ok(a / b);

const result = divide(10, 2);
if (result.isOk()) {
  console.log(result.value); // 5
}

// Pattern matching
result.match({
  ok: (value) => `Result: ${value}`,
  err: (error) => `Error: ${error}`,
});
```

### Pipe Style (Dual API)

```typescript
import { pipe, ok, map, filter, tap, getOrElse } from '@deessejs/fp';

const value = pipe(
  ok(10),
  map(x => x * 2),
  filter(x => x > 0, () => new Error('Not positive')),
  tap(x => console.log(x)),
  getOrElse(0),
);
```

### Maybe for Optional Values

```typescript
import { Maybe, some, none, pipe, map, filter, getOrElse } from '@deessejs/fp';

// Safe property access
const getCity = (user: User | null): Maybe<string> =>
  user?.address?.city ? some(user.address.city) : none;

// With pipe
const formattedCity = pipe(
  getCity(user),
  map(city => city.toUpperCase()),
  filter(city => city.length > 3),
  getOrElse('Unknown'),
);
```

### Result with @deessejs/errors

```typescript
import { Result, ok, err, mapError } from '@deessejs/fp';
import { error, is } from '@deessejs/errors';

const ValidationError = error({
  name: 'ValidationError',
  message: 'Validation failed: {reason}',
});

const NotFoundError = error({
  name: 'NotFoundError',
  message: 'User "{id}" not found',
});

async function getUser(id: string): Promise<Result<User, ValidationError | NotFoundError>> {
  if (!id) {
    return err(ValidationError({ reason: 'ID is required' }));
  }

  const user = await db.findById(id);
  if (!user) {
    return err(NotFoundError({ id }));
  }

  return ok(user);
}

// Usage
const result = await getUser('123');
result.match({
  ok: (user) => console.log(user),
  err: (e) => {
    if (is(e, ValidationError)) {
      console.log(e.fields.reason);
    } else if (is(e, NotFoundError)) {
      console.log(`User ${e.fields.id} not found`);
    }
  },
});
```
