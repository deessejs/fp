# v1.1.0 — Async, Collections & Advanced Patterns

**Status:** Planned  
**Type:** Minor Changes  
**SemVer:** 1.1.0  
**Depends on:** v1.0.0

## Overview

v1.1.0 adds async utilities, collection types, generator composition, and serialization. These are practical patterns for real-world applications.

## Changes

### Minor Changes

- Add `sleep()` for promise-based delay
- Add `retry()` with configurable policies
- Add `timeout()` for operation cancellation
- Add `jitter()` for random delay
- Add delay strategies: `exponential`, `linear`, `constant`
- Add `queue()` for async job processing
- Add `Context<T>` for config precedence
- Add `Sequence<T>` for lazy iteration
- Add `Collection<T>` for set-like operations
- Add AsyncIterator utilities: `collect`, `first`, `last`, `mapAsync`, `filterAsync`
- Add `gen()` for generator composition
- Add `await()` for yieldable Promise<Result>
- Add `serialize()` / `deserialize()` for RPC
- Add `partition()` for splitting Results

## What's Included

### Async Utilities

```typescript
function sleep(ms: number): Promise<void>;

function retry<T>(config: RetryConfig): (thunk: () => Promise<T>) => Promise<T>;

function timeout<T>(ms: number, thunk: () => Promise<T>): Promise<T>;

function jitter(maxMs: number, minMs?: number): number;

type DelayStrategy = typeof exponential | typeof linear | typeof constant;
```

### Queue

```typescript
interface Queue<T> {
  add(job: () => Promise<T>, options?: { priority?: number }): void;
  flush(): Promise<T[]>;
  clear(): void;
  readonly size: number;
}

function queue<T>(config?: QueueConfig): Queue<T>;
```

### Collection Types

```typescript
// Context - config precedence
interface Context<T> {
  get(key: string): Maybe<T>;
  get(key: string, fallback: T): T;
  has(key: string): boolean;
  push(map: Record<string, T>): Context<T>;
  scope(overrides: Record<string, T>): Context<T>;
}

function context<T>(): Context<T>;

// Sequence - lazy iteration
interface Sequence<T> {
  map<B>(fn: (value: T) => B): Sequence<B>;
  filter(predicate: (value: T) => boolean): Sequence<T>;
  take(count: number): Sequence<T>;
  toArray(): T[];
}

namespace sequence {
  function range(start: number, end: number): Sequence<number>;
  function from<T>(iterable: Iterable<T>): Sequence<T>;
  function unfold<A, B>(seed: A, fn: (a: A) => [B, A] | null): Sequence<B>;
}

// Collection - set operations
interface Collection<T> {
  union(other: Collection<T>): Collection<T>;
  intersection(other: Collection<T>): Collection<T>;
  difference(other: Collection<T>): Collection<T>;
  toArray(): T[];
}

function collection<T>(items: T[]): Collection<T>;
```

### AsyncIterator Utilities

```typescript
async function collect<T>(
  iterable: AsyncIterable<T>,
  options?: { limit?: number }
): Promise<T[]>;

async function first<T>(iterable: AsyncIterable<T>): Promise<Maybe<T>>;
async function last<T>(iterable: AsyncIterable<T>): Promise<Maybe<T>>;
async function mapAsync<T, B>(
  iterable: AsyncIterable<T>,
  fn: (item: T) => B | Promise<B>
): Promise<B[]>;
async function filterAsync<T>(
  iterable: AsyncIterable<T>,
  predicate: (item: T) => boolean | Promise<boolean>
): Promise<T[]>;
```

### Generator Composition

```typescript
function gen<R extends AnyResult>(
  body: () => AsyncGenerator<Result<unknown, unknown>, R, unknown>
): Promise<Result<InferOk<R>, InferErr<R>>>;

function await<T, E>(
  promise: Promise<Result<T, E>>
): AsyncGenerator<Err<never, E>, T, unknown>;
```

### Serialization

```typescript
type SerializedOk<T> = { status: "ok"; value: T; };
type SerializedErr<E> = { status: "error"; error: E; };
type SerializedResult<T, E> = SerializedOk<T> | SerializedErr<E>;

function serialize<T, E>(result: Result<T, E>): SerializedResult<T, E>;
function deserialize<T, E>(value: unknown): Result<T, E | DeserializationError>;
function partition<T, E>(results: readonly Result<T, E>[]): [T[], E[]];
```

## Usage Example

```typescript
import { sleep, retry, timeout, queue, exponential } from '@deessejs/fp';
import { context, sequence, collection } from '@deessejs/fp';
import { gen, serialize, deserialize } from '@deessejs/fp';

// Retry with exponential backoff
const fetchWithRetry = retry({
  attempts: 3,
  delay: exponential(100),
})(() => fetch(url));

// Job queue
const uploadQueue = queue({ concurrency: 2 });
uploadQueue.add(() => uploadFile(file));
await uploadQueue.flush();

// Context for config
const config = context()
  .push({ port: 3000 })
  .push({ port: parseInt(process.env.PORT ?? '') });

// Lazy sequence
const fibonacci = sequence.unfold(
  [0, 1] as [number, number],
  ([a, b]) => [a, [b, a + b]] as [number, [number, number]]
);
const first10 = fibonacci.take(10).toArray();

// Generator composition
async function registerUser(input: RegisterInput) {
  return gen(async function* () {
    const valid = yield* validateInput(input);
    const user = yield* await tryPromise(() => db.users.create(valid));
    return ok(user);
  });
}

// Serialization
const serialized = serialize(result);
// { status: "ok", value: {...} }
```
