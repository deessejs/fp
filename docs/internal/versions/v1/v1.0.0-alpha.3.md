# v1.0.0-alpha.3 — Try Utilities

**Status:** Planned  
**Type:** Minor Changes  
**SemVer:** 1.0.0-alpha.3  
**Depends on:** v1.0.0-alpha.1

## Overview

The third alpha release adds `try_` and `tryPromise` for wrapping operations that may throw. This makes the "never let exceptions escape" promise credible.

## Changes

### Minor Changes

- Add `try_()` for sync exception handling
- Add `tryPromise()` for async exception handling

## What's Included

### try_

Wraps a synchronous function that may throw.

```typescript
// Simple form
function try_<A>(thunk: () => A): Result<A, UnhandledException>;

// With custom error handler
function try_<A, E>(options: {
  try: () => A;
  catch: (cause: unknown) => E;
}): Result<A, E>;
```

### tryPromise

Wraps an async function that may reject.

```typescript
// Simple form
function tryPromise<A>(
  thunk: () => Promise<A>
): Promise<Result<A, UnhandledException>>;

// With custom error handler
function tryPromise<A, E>(options: {
  try: () => Promise<A>;
  catch: (cause: unknown) => E | Promise<E>;
}): Promise<Result<A, E>>;

// With retry
function tryPromise<A, E>(
  thunk: () => Promise<A>,
  config: RetryConfig<E>
): Promise<Result<A, E>>;
```

### UnhandledException

```typescript
interface UnhandledException {
  readonly name: 'UnhandledException';
  readonly cause: unknown;
}
```

## Usage Example

```typescript
import { try_, tryPromise, Result, ok, err } from '@deessejs/fp';
import { error } from '@deessejs/errors';

const ParseError = error({
  name: 'ParseError',
  message: 'Parse error: {cause}',
});

// Sync wrapping
const parsed = try_(() => JSON.parse(input));
// Result<unknown, UnhandledException>

// With custom error
const parsed = try_({
  try: () => JSON.parse(input) as unknown,
  catch: (e) => ParseError({ cause: e }),
});
// Result<unknown, ParseError>

// Async wrapping
const user = await tryPromise(() => fetch('/api/user').then(r => r.json()));
// Promise<Result<User, UnhandledException>>

// With custom error
const user = await tryPromise({
  try: () => fetch('/api/user').then(r => r.json()),
  catch: (e) => ApiError({ cause: e }),
});
// Promise<Result<User, ApiError>>
```
