# v1.0.0-beta.1 — Validation + Error Patterns

**Status:** Planned  
**Type:** Minor Changes  
**SemVer:** 1.0.0-beta.1  
**Depends on:** v1.0.0-alpha.3

## Overview

The first beta release adds predicates for validation and advanced error patterns for production-ready error handling.

## Changes

### Minor Changes

- Add `Predicate` and `Refinement` types
- Add logical combinators: `not`, `and`, `or`, `xor`
- Add constant utilities: `constTrue`, `constFalse`, `constNull`, `constUndefined`, `constVoid`
- Add `absurd()` for exhaustive checks
- Add `increment()` / `decrement()` utilities
- Add `attempt()` for configurable error handling
- Add `sanitizeMessage()` for client-safe errors
- Add `toClientSafe()` for error normalization
- Add `classifyError()` for retry classification
- Add `withReporting()` for error reporting

## What's Included

### Predicate Types

```typescript
type Predicate<A> = (value: A) => boolean;
type Refinement<A, B extends A> = (value: A) => value is B;
```

### Logical Combinators

```typescript
function not<A>(predicate: Predicate<A>): Predicate<A>;
function and<A>(a: Predicate<A>, b: Predicate<A>): Predicate<A>;
function and<A>(...predicates: Predicate<A>[]): Predicate<A>;
function or<A>(a: Predicate<A>, b: Predicate<A>): Predicate<A>;
function or<A>(...predicates: Predicate<A>[]): Predicate<A>;
function xor<A>(a: Predicate<A>, b: Predicate<A>): Predicate<A>;
```

### Constants

```typescript
function constTrue<A>(): Predicate<A>;
function constFalse<A>(): Predicate<A>;
function constNull(): null;
function constUndefined(): undefined;
function constVoid(): void;
function absurd<A>(_: never): A;
function increment(n: number): number;
function decrement(n: number): number;
```

### Attempt (Advanced)

```typescript
interface AttemptConfig<T> {
  try: () => T | Promise<T>;
  client?: boolean;
  retry?: RetryConfig<unknown>;
  normalize?: (e: unknown) => unknown;
}

interface Attempt<T> {
  execute(): Promise<Result<T, unknown>>;
  clientSafe(): Promise<Result<T, NormalizedError>>;
}

function attempt<T>(config: AttemptConfig<T>): Attempt<T>;
```

### Error Normalization

```typescript
interface NormalizedError {
  code: string;
  message: string;
  status: number;
  public: boolean;
}

type ErrorNormalizer<E> = (e: E) => NormalizedError;

function sanitizeMessage(message: string): string;

function toClientSafe<T>(
  operation: () => Promise<T>,
  normalizer: ErrorNormalizer<unknown>
): Promise<Result<T, NormalizedError>>;
```

### Error Classification

```typescript
type ErrorClassification = 'retryable' | 'non-retryable';

interface ClassificationRule {
  error: ErrorFactory;
  classification: ErrorClassification;
}

function classifyError(
  e: unknown,
  rules: ClassificationRule[]
): ErrorClassification;
```

### Error Reporting

```typescript
interface ErrorReporter {
  report(error: unknown, context: ErrorContext): void;
}

interface ErrorContext {
  timestamp: number;
  operation: string;
  metadata?: Record<string, unknown>;
}

function withReporting<T>(
  operation: () => Promise<T>,
  operationName: string,
  reporter: ErrorReporter,
  metadata?: Record<string, unknown>
): Promise<Result<T, ReportableError>>;
```

## Usage Example

```typescript
import { Predicate, not, and, or, attempt, toClientSafe } from '@deessejs/fp';
import { error } from '@deessejs/errors';

// Predicates for validation
const isEmail: Predicate<string> = s => s.includes('@');
const isNonEmpty: Predicate<string> = s => s.length > 0;
const minLength = (min: number): Predicate<string> => s => s.length >= min;

const isValidPassword = and(
  minLength(8),
  (s) => /[A-Z]/.test(s),
  (s) => /[0-9]/.test(s),
);

// Client-safe error handling
const normalizers = {
  DatabaseError: () => ({
    code: 'DB_ERROR',
    message: 'Service temporarily unavailable',
    status: 503,
    public: true,
  }),
};

const result = await toClientSafe(
  () => fetchUser(id),
  (e) => normalizers[e.constructor.name]?.() ?? {
    code: 'UNKNOWN',
    message: 'An error occurred',
    status: 500,
    public: true,
  }
);

// Attempt with retry
const attempt_ = attempt({
  try: () => fetchData(),
  retry: { attempts: 3, delay: exponential(100) },
  client: true,
});

const result = await attempt_.clientSafe();
```
